cmake_minimum_required(VERSION 3.16)
project(FantasyQuest VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32 AND EXISTS "${CMAKE_SOURCE_DIR}/.tools/SFML-2.6.1")
    list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/.tools/SFML-2.6.1")
endif()
find_package(SFML 2.6 COMPONENTS graphics window audio system REQUIRED)
add_executable(FantasyQuest src/Main.cpp src/Model.cpp src/Frontend.cpp src/GameViews.cpp src/Application.cpp)
target_link_libraries(FantasyQuest PRIVATE sfml-graphics sfml-window sfml-audio sfml-system)
target_include_directories(FantasyQuest PRIVATE src)
if(MSVC)
    target_compile_options(FantasyQuest PRIVATE /W4 /utf-8)
else()
    target_compile_options(FantasyQuest PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_command(TARGET FantasyQuest POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:FantasyQuest>/assets")
if(WIN32)
    foreach(library graphics window audio system)
        add_custom_command(TARGET FantasyQuest POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:sfml-${library}>" "$<TARGET_FILE_DIR:FantasyQuest>")
    endforeach()
endif()
enable_testing()
add_executable(ModelTests tests/ModelTests.cpp tests/RenderStubs.cpp src/Model.cpp)
target_include_directories(ModelTests PRIVATE src)
add_test(NAME CartridgeRules COMMAND ModelTests)
set_tests_properties(CartridgeRules PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
